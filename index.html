-- Crear detalle de cada pedido CREATE TABLE IF NOT EXISTS public."PedidoDetalles" ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, pedido_id BIGINT NOT NULL REFERENCES public."Pedidos"(id) ON DELETE CASCADE, producto_id BIGINT NOT NULL, nombre TEXT NOT NULL, precio NUMERIC(12,2) NOT NULL, cantidad INTEGER NOT NULL CHECK (cantidad > 0), subtotal NUMERIC(12,2) NOT NULL ); ALTER TABLE public."Pedidos" ENABLE ROW LEVEL SECURITY; ALTER TABLE public."PedidoDetalles" ENABLE ROW LEVEL SECURITY; -- Registrar pedido y descontar inventario de forma segura CREATE OR REPLACE FUNCTION public.confirmar_pedido(p_items JSONB) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_item RECORD; END IF; IF EXISTS ( SELECT 1 FROM jsonb_to_recordset(p_items) AS x(id BIGINT, cantidad INTEGER) WHERE x.id IS NULL OR x.cantidad IS NULL OR x.cantidad <= 0 ) THEN RAISE EXCEPTION 'Producto o cantidad no válida.'; END IF; FOR v_item IN SELECT x.id, SUM(x.cantidad)::INTEGER AS cantidad FROM jsonb_to_recordset(p_items) AS x(id BIGINT, cantidad INTEGER) GROUP BY x.id ORDER BY x.id LOOP SELECT p."Nombre", p."Precio", p."Stock", p."Activo" INTO v_nombre, v_precio, v_stock, v_activo FROM public."Productos" p WHERE p.id = v_item.id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Producto ID % no encontrado.', v_item.id; END IF; IF v_activo IS DISTINCT FROM TRUE THEN RAISE EXCEPTION 'El producto "%" no está disponible.', v_nombre; END IF; IF v_stock IS NULL OR v_stock < v_item.cantidad THEN RAISE EXCEPTION 'Existencias insuficientes para "%". Disponibles: %.', v_nombre, COALESCE(v_stock, 0); END IF; IF v_precio IS NULL OR v_precio <= 0 THEN RAISE EXCEPTION 'El producto "%" no tiene precio válido.', v_nombre; END IF; v_total := v_total + v_precio * v_item.cantidad; v_detalles := v_detalles || jsonb_build_array( jsonb_build_object( 'producto_id', v_item.id, 'nombre', v_nombre, 'precio', v_precio, 'cantidad', v_item.cantidad, 'subtotal', v_precio * v_item.cantidad ) ); END LOOP; INSERT INTO public."Pedidos"(total) VALUES (v_total) RETURNING id INTO v_pedido_id; FOR v_item IN SELECT x.id, SUM(x.cantidad)::INTEGER AS cantidad FROM jsonb_to_recordset(p_items) AS x(id BIGINT, cantidad INTEGER) GROUP BY x.id ORDER BY x.id LOOP UPDATE public."Productos" SET "Stock" = "Stock" - v_item.cantidad WHERE id = v_item.id; INSERT INTO public."PedidoDetalles"( pedido_id, producto_id, nombre, precio, cantidad, subtotal ) SELECT v_pedido_id, (d->>'producto_id')::BIGINT, d->>'nombre', (d->>'precio')::NUMERIC, (d->>'cantidad')::INTEGER, (d->>'subtotal')::NUMERIC FROM jsonb_array_elements(v_detalles) AS x(d) WHERE (d->>'producto_id')::BIGINT = v_item.id; END LOOP; RETURN jsonb_build_object( 'pedido_id', v_pedido_id, 'total', v_total, 'detalles', v_detalles ); END; $$; REVOKE ALL ON FUNCTION public.confirmar_pedido(JSONB) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.confirmar_pedido(JSONB) TO anon, authenticated;
